>> Group by, Having Query
    GROUP BY:
        -- department wise group (cse,eee,me,ete,etc)
        -- country wise (bd, ind, aus)
        "
            SELECT MAX(SALARY)
            FROM EMPLOYEES
            GROUP BY DEPARTMENT_ID;

            SELECT DEPARTMENT_ID, COUNT(*)
            FROM EMPLOYEES
            GROUP BY DEPARTMENT_ID;


            SELECT *
            FROM EMPLOYEES
            WHERE DEPARTMENT_ID = 20
        "
    HAVING :
        -- Group function ar opore applied condition
        (WHERE is different condition)
        "
            SELECT DEPARTMENT_ID, AVG(SALARY) AS SAL
            FROM EMPLOYEES
            WHERE DEPARTMENT_ID != 20
            GROUP BY DEPARTMENT_ID
            HAVING SAL > 5000;

            SELECT DEPARTMENT_ID, AVG(SALARY)
            FROM EMPLOYEES
            WHERE DEPARTMENT_ID != 20
            GROUP BY DEPARTMENT_ID
            HAVING AVG(SALARY) > 5000;

            SELECT DEPARTMENT_ID, MAX(SALARY)
            FROM EMPLOYEES
            WHERE DEPARTMENT_ID != 20
            GROUP BY DEPARTMENT_ID
            HAVING MAX(SALARY) > 5000;

            SELECT DEPARTMENT_ID, MAX(SALARY)
            FROM EMPLOYEES
            WHERE DEPARTMENT_ID != 20
            GROUP BY DEPARTMENT_ID;
        "
